## Usage

```swift
FLXFormField {
    FLXFieldLabel("Email") {
        FLXTextField("teammate@company.com", text: $email)
    }
}
```

## Composition

`FLXFormField` has no `label`/`description`/`error` parameters. Compose [`FLXFieldLabel`](/docs/components/mac/field-label), the control, `FLXFormFieldDescription`, and `FLXFormFieldError` as children:

<ComponentPreviewCode
  code={`FLXFormField {
    FLXFieldLabel("Username") {
        FLXTextField("Username", text: $username)
    }
    FLXFormFieldDescription("Visible to other members.")
    if username.count < 3 {
        FLXFormFieldError("Must be at least 3 characters.")
    }
}
`}
  image={{ light: "/images/components/mac/form-field/form-field-error.webp", dark: "/images/components/mac/form-field/form-field-error-dark.webp" }}
/>

## Custom styling

`FLXFormFieldStyle`, `FLXFormFieldDescriptionStyle`, and `FLXFormFieldErrorStyle` are FLX protocols. Override each independently via `.flxFormFieldStyle(_:)`, `.flxFormFieldDescriptionStyle(_:)`, and `.flxFormFieldErrorStyle(_:)`:

<ComponentPreviewCode
  code={`struct FLXWarningFormFieldErrorStyle: FLXFormFieldErrorStyle {
    func makeBody(configuration: Configuration) -> some View {
        Label {
            configuration.text
        } icon: {
            Image(systemName: "exclamationmark.triangle.fill")
        }
            .flxFont(.footnote)
            .foregroundStyle(.orange)
    }
}

VStack(spacing: FLXSpacing.lg) {
    FLXFormField {
        FLXFieldLabel("Email") {
            FLXTextField("Email", text: $email)
        }
        FLXFormFieldError("Something went wrong.")
    }
    FLXFormField {
        FLXFieldLabel("Email") {
            FLXTextField("Email", text: $email)
        }
        FLXFormFieldError("Something went wrong.")
    }
    .flxFormFieldErrorStyle(FLXWarningFormFieldErrorStyle())
}
`}
  image={{ light: "/images/components/mac/form-field/form-field-custom.webp", dark: "/images/components/mac/form-field/form-field-custom-dark.webp" }}
  collapsible
/>
